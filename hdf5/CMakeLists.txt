cmake_minimum_required(VERSION 3.18)
project(radelta_hdf5 VERSION 0.7.3 LANGUAGES C)
include(CTest)
find_package(HDF5 1.8.15 REQUIRED COMPONENTS C)
if(TARGET HDF5::HDF5)
    set(RADELTA_HDF5_LIBRARIES HDF5::HDF5)
else()
    set(RADELTA_HDF5_LIBRARIES ${HDF5_LIBRARIES})
endif()
find_program(CARGO_EXECUTABLE cargo REQUIRED)
set(RADELTA_HDF5_FILTER_ID 65024 CACHE STRING "Private HDF5 filter ID (not globally registered)")
if(RADELTA_HDF5_FILTER_ID LESS 32768 OR RADELTA_HDF5_FILTER_ID GREATER 65535)
    message(FATAL_ERROR "Use a private filter ID in 32768..65535")
endif()
get_filename_component(RADELTA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(RUST_TARGET "${CMAKE_CURRENT_BINARY_DIR}/cargo")
if(WIN32)
    set(RUST_LIBRARY "${RUST_TARGET}/release/radelta_native.lib")
else()
    set(RUST_LIBRARY "${RUST_TARGET}/release/libradelta_native.a")
endif()
file(GLOB RUST_SOURCES CONFIGURE_DEPENDS "${RADELTA_ROOT}/native/src/*.rs")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CARGO_EXECUTABLE}" build --release --locked --lib
        --manifest-path "${RADELTA_ROOT}/native/Cargo.toml" --target-dir "${RUST_TARGET}"
    DEPENDS ${RUST_SOURCES} "${RADELTA_ROOT}/native/Cargo.toml" "${RADELTA_ROOT}/native/Cargo.lock"
    WORKING_DIRECTORY "${RADELTA_ROOT}" VERBATIM)
add_custom_target(radelta_codec DEPENDS "${RUST_LIBRARY}")
add_library(H5Zradelta MODULE src/H5Zradelta.c)
add_dependencies(H5Zradelta radelta_codec)
set_target_properties(H5Zradelta PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
    WINDOWS_EXPORT_ALL_SYMBOLS YES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plugins")
if(APPLE)
    set_target_properties(H5Zradelta PROPERTIES SUFFIX ".dylib"
        INSTALL_RPATH "@loader_path/../..")
elseif(UNIX)
    set_target_properties(H5Zradelta PROPERTIES INSTALL_RPATH "$ORIGIN/../..")
endif()
target_compile_definitions(H5Zradelta PRIVATE RADELTA_HDF5_FILTER_ID=${RADELTA_HDF5_FILTER_ID})
target_include_directories(H5Zradelta PRIVATE ${HDF5_INCLUDE_DIRS} "${RADELTA_ROOT}/native/include")
target_link_libraries(H5Zradelta PRIVATE "${RUST_LIBRARY}" ${RADELTA_HDF5_LIBRARIES})
target_compile_options(H5Zradelta PRIVATE ${HDF5_DEFINITIONS})
if(WIN32)
    target_link_libraries(H5Zradelta PRIVATE ws2_32 userenv bcrypt ntdll)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(H5Zradelta PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
if(MSVC)
    target_compile_options(H5Zradelta PRIVATE /W4)
else()
    target_compile_options(H5Zradelta PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS H5Zradelta LIBRARY DESTINATION lib/hdf5/plugins RUNTIME DESTINATION lib/hdf5/plugins)

# Read-only filter for existing native HDF5 readers, including Fiji/JHDF5.
# Headers define the stable filter-plugin ABI; no HDF5 library is linked.
add_library(H5Zradelta_fiji MODULE src/H5Zradelta.c)
add_dependencies(H5Zradelta_fiji radelta_codec)
set_target_properties(H5Zradelta_fiji PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES
    OUTPUT_NAME H5Zradelta WINDOWS_EXPORT_ALL_SYMBOLS YES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fiji-plugins"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fiji-plugins"
    MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
if(APPLE)
    set_target_properties(H5Zradelta_fiji PROPERTIES SUFFIX ".dylib")
endif()
target_compile_definitions(H5Zradelta_fiji PRIVATE RADELTA_HDF5_DECODE_ONLY=1
    RADELTA_HDF5_FILTER_ID=${RADELTA_HDF5_FILTER_ID})
target_include_directories(H5Zradelta_fiji PRIVATE ${HDF5_INCLUDE_DIRS} "${RADELTA_ROOT}/native/include")
target_link_libraries(H5Zradelta_fiji PRIVATE "${RUST_LIBRARY}")
if(WIN32)
    target_link_libraries(H5Zradelta_fiji PRIVATE ws2_32 userenv bcrypt ntdll)
else()
    target_link_libraries(H5Zradelta_fiji PRIVATE Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
if(MSVC)
    target_compile_options(H5Zradelta_fiji PRIVATE /W4)
else()
    target_compile_options(H5Zradelta_fiji PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS H5Zradelta_fiji LIBRARY DESTINATION lib/hdf5/fiji-plugins
    RUNTIME DESTINATION lib/hdf5/fiji-plugins)
if(BUILD_TESTING)
    add_executable(hdf5_roundtrip tests/roundtrip.c)
    target_include_directories(hdf5_roundtrip PRIVATE ${HDF5_INCLUDE_DIRS})
    target_link_libraries(hdf5_roundtrip PRIVATE ${RADELTA_HDF5_LIBRARIES})
    target_compile_options(hdf5_roundtrip PRIVATE ${HDF5_DEFINITIONS})
    target_compile_definitions(hdf5_roundtrip PRIVATE RADELTA_HDF5_FILTER_ID=${RADELTA_HDF5_FILTER_ID})
    add_test(NAME hdf5_roundtrip COMMAND hdf5_roundtrip)
    set_tests_properties(hdf5_roundtrip PROPERTIES ENVIRONMENT "HDF5_PLUGIN_PATH=$<TARGET_FILE_DIR:H5Zradelta>")
    if(APPLE)
        set_target_properties(hdf5_roundtrip PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    elseif(UNIX)
        set_target_properties(hdf5_roundtrip PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    install(TARGETS hdf5_roundtrip RUNTIME DESTINATION bin)

    add_executable(hdf5_fiji_fixture tests/fiji_fixture.c)
    set_target_properties(hdf5_fiji_fixture PROPERTIES OUTPUT_NAME radelta-hdf5-fiji-fixture
        C_STANDARD 99 C_STANDARD_REQUIRED YES)
    target_include_directories(hdf5_fiji_fixture PRIVATE ${HDF5_INCLUDE_DIRS})
    target_link_libraries(hdf5_fiji_fixture PRIVATE ${RADELTA_HDF5_LIBRARIES})
    target_compile_options(hdf5_fiji_fixture PRIVATE ${HDF5_DEFINITIONS})
    target_compile_definitions(hdf5_fiji_fixture PRIVATE RADELTA_HDF5_FILTER_ID=${RADELTA_HDF5_FILTER_ID})
    if(APPLE)
        set_target_properties(hdf5_fiji_fixture PROPERTIES INSTALL_RPATH "@loader_path/../lib")
    elseif(UNIX)
        set_target_properties(hdf5_fiji_fixture PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    install(TARGETS hdf5_fiji_fixture RUNTIME DESTINATION bin)
endif()
